# Stage 1: build
FROM node:20-bullseye AS build

# Install build tools and curl
RUN apt-get update && apt-get install -y curl git build-essential ca-certificates && rm -rf /var/lib/apt/lists/*

# Install bun (used by your repo's build script)
RUN curl -fsSL https://bun.sh/install | bash
ENV BUN_INSTALL=/root/.bun
ENV PATH=/root/.bun/bin:/usr/local/bin:$PATH

WORKDIR /app
# Copy whole repo (monorepo)
COPY . .

# Try bun install (uses bun.lockb if present), ignore failure as fallback will handle
RUN bun install || true

# Build the common package (repo's backend build expects it)
RUN cd common && (bun run build || (npm install && npm run build) || npx tsc)

# Build backend
RUN cd backend && npm install && npx tsc

# Stage 2: runtime
FROM node:20-bullseye AS runtime

WORKDIR /app
# Copy compiled backend
COPY --from=build /app/backend/dist ./dist
# Copy node_modules for backend runtime
COPY --from=build /app/backend/node_modules ./node_modules
# Copy other runtime assets if used
COPY --from=build /app/backend/public ./public || true

ENV NODE_ENV=production
EXPOSE 3000
# Adjust path if your backend's entry file is different
CMD ["node", "dist/index.js"]
